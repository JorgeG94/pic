   function splitmix64_next_real_dp(gen) result(r)
      !! Uniform `real(dp)` in [0, 1) from a SplitMix64 generator.
      type(splitmix64_t), intent(inout) :: gen
      real(dp) :: r

      r = real_from_word(gen%next_u64())
   end function splitmix64_next_real_dp
