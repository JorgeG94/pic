   function pcg32_next_real_dp(gen) result(r)
      !! Uniform `real(dp)` in [0, 1) from a PCG32 generator. Consumes two
      !! 32-bit outputs.
      type(pcg32_t), intent(inout) :: gen
      real(dp) :: r

      r = real_from_word(gen%next_u64())
   end function pcg32_next_real_dp
