   pure subroutine update_rsp_2(this, key)
      class(array_hash_t), intent(inout) :: this
      real(sp), intent(in) :: key(:, :)

      integer(default_int) :: i

      do i = 1_default_int, size(key, 2, kind=default_int)
         call this%update(key(:, i))
      end do
