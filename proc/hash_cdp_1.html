   pure function hash_cdp_1(key) result(hash_value)
      complex(dp), intent(in) :: key(:)
      integer(int32) :: hash_value

      type(array_hash_t) :: state

      call state%update(key)
      hash_value = state%digest()
   end function hash_cdp_1
